Help

#include "oslib/help.h"

help_flags

typedef bits help_flags;
Values:
help_DISABLE
help_READ_CONFIGURATION

help_message_request

typedef struct
   {  os_coord pos;
      wimp_mouse_state buttons;
      wimp_w w;
      wimp_i i;
   }
   help_message_request;

message type to request help from an application

Fields: pos - mouse coordinates buttons - mouse button state w - window handle i - icon handle

help_message_reply

typedef struct
   {  char reply [236];
   }
   help_message_reply;

message type to reply with help text

Fields: reply - help message, 0-terminated

help_message_enable

typedef struct
   {  help_flags flags;
   }
   help_message_enable;

message type to switch help on and off - RISC OS 4+

ExtEdit

#include "oslib/extedit.h"

extedit_data_type

typedef bits extedit_data_type;
Values:
extedit_FILE_TYPE
extedit_SUB_TYPE

extedit_job_handle

typedef bits extedit_job_handle;
Values:
extedit_CLIENT_HANDLE
extedit_EDITOR_HANDLE

extedit_flags

typedef bits extedit_flags;
Values:
extedit_CONTINUE_EDITING
extedit_SELECTION_ONLY
extedit_READ_ONLY
extedit_IMMEDIATE_EXECUTION
extedit_ADJUST_SELECTION

extedit_message_rq

typedef struct
   {  extedit_data_type data_type;
      extedit_job_handle job_handle;
      extedit_flags flags;
      char parent_name [20];
      char leaf_name [204];
   }
   extedit_message_rq;

extedit_message_ack

typedef struct
   {  extedit_data_type data_type;
      extedit_job_handle job_handle;
      extedit_flags flags;
   }
   extedit_message_ack;

extedit_message_return

typedef struct
   {  extedit_data_type data_type;
      extedit_job_handle job_handle;
      extedit_flags flags;
   }
   extedit_message_return;

extedit_message_abort

typedef struct
   {  byte reserved [4];
      extedit_job_handle job_handle;
   }
   extedit_message_abort;

extedit_message_data_save

typedef struct
   {  extedit_job_handle job_handle;
      byte reserved [12];
      int est_size;
      bits file_type;
      char file_name [212];
   }
   extedit_message_data_save;

extedit_message_cursor

typedef struct
   {  byte reserved [4];
      extedit_job_handle job_handle;
      extedit_flags flags;
      int new_cursor_pos;
      int new_sos_pos;
      int new_eos_pos;
      int old_cursor_pos;
      int old_sos_pos;
      int old_eos_pos;
   }
   extedit_message_cursor;
